cmake_minimum_required(VERSION 3.20)
project(tq_classic_foundation VERSION 1.3.0 LANGUAGES CXX)
include(CTest)
add_library(tq_classic_rules src/classic_rules.cpp)
target_include_directories(tq_classic_rules PUBLIC src)
target_compile_features(tq_classic_rules PUBLIC cxx_std_17)
add_executable(tq_classic_scaffold src/main.cpp)
target_link_libraries(tq_classic_scaffold PRIVATE tq_classic_rules)
if(BUILD_TESTING)
    add_executable(tq_classic_rules_test tests/classic_rules_test.cpp)
    target_link_libraries(tq_classic_rules_test PRIVATE tq_classic_rules)
    add_test(NAME classic_rules COMMAND tq_classic_rules_test)
endif()
foreach(target tq_classic_rules tq_classic_scaffold)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive-)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endforeach()
find_package(SQLite3 QUIET)
if(SQLite3_FOUND)
    add_library(tq_daily_claim_store src/daily_claim_store.cpp)
    target_link_libraries(tq_daily_claim_store PUBLIC tq_classic_rules SQLite::SQLite3)
    target_compile_features(tq_daily_claim_store PUBLIC cxx_std_17)
    if(BUILD_TESTING)
        add_executable(tq_daily_claim_store_test tests/daily_claim_store_test.cpp)
        target_link_libraries(tq_daily_claim_store_test PRIVATE tq_daily_claim_store)
        add_test(NAME daily_claim_store COMMAND tq_daily_claim_store_test)
    endif()
else()
    message(STATUS "SQLite3 not found: persistent daily claims module not built")
endif()

    add_test(NAME diagnostic_rejects_invalid_port COMMAND tq_tcp_diagnostic 0)
    set_tests_properties(diagnostic_rejects_invalid_port PROPERTIES WILL_FAIL TRUE)
endif()
